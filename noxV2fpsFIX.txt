    if activeSlider then
        local s = activeSlider
        local mouse = UserInputService:GetMouseLocation()
        local barAbs = s.bar.AbsolutePosition
        local barSize = s.bar.AbsoluteSize
        local p = math.clamp((mouse.X - barAbs.X) / barSize.X, 0, 1)
        s.ref.value = s.minVal + p * (s.maxVal - s.minVal)
        s.ref.value = math.floor(s.ref.value / s.step) * s.step
        local pct = math.clamp((s.ref.value - s.minVal) / (s.maxVal - s.minVal), 0, 1)
        s.knob.Position = UDim2.new(pct, -7, 0.5, -7)
        s.fill.Size = UDim2.new(pct, 0, 1, 0)
        s.valueLabel.Text = tostring(math.floor(s.ref.value))
        s.callback(s.ref.value)
    end
end)

local clearESP, clearChams, clearOverlay, restoreHitboxes, updateESP, updateESPHp, updateTheme, recalcPositions

local function isTeammate(targetPlayer)
    if not espTeamCheck then return false end
    if not LocalPlayer.Team or not targetPlayer.Team then return false end
    return LocalPlayer.Team == targetPlayer.Team
end

clearESP = function()
    for _, data in pairs(espObjects) do
        if data.billboard and data.billboard.Parent then data.billboard:Destroy() end
        if data.highlight and data.highlight.Parent then data.highlight:Destroy() end
    end
    espObjects = {}
    for _, lines in pairs(box3DObjects) do
        for _, line in ipairs(lines) do if line then pcall(function() line:Remove() end) end end
    end
    box3DObjects = {}
    for _, tracer in pairs(tracerLines) do if tracer then pcall(function() tracer:Remove() end) end end
    tracerLines = {}
end

clearChams = function()
    for _, h in pairs(chamsCache) do
        if h and h.Parent then h:Destroy() end
    end
    chamsCache = {}
end

clearOverlay = function()
    for _, el in pairs(overlayElements) do
        if el and el.Parent then el:Destroy() end
    end
    overlayElements = {}
end

restoreHitboxes = function()
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            for _, part in pairs(plr.Character:GetChildren()) do
                if part:IsA("BasePart") then
                    local key = plr.Name .. part.Name
                    if originalSizes[key] then
                        part.Size = originalSizes[key]
                        originalSizes[key] = nil
                    end
                end
            end
        end
    end
end

local function clearPlayerESP(targetPlayer)
    if espObjects[targetPlayer] then
        if espObjects[targetPlayer].billboard then pcall(function() espObjects[targetPlayer].billboard:Destroy() end) end
        if espObjects[targetPlayer].highlight then pcall(function() espObjects[targetPlayer].highlight:Destroy() end) end
        espObjects[targetPlayer] = nil
    end
    if tracerLines[targetPlayer] then pcall(function() tracerLines[targetPlayer]:Remove() end); tracerLines[targetPlayer] = nil end
    if box3DObjects[targetPlayer] then
        for _, line in ipairs(box3DObjects[targetPlayer]) do if line then pcall(function() line:Remove() end) end end
        box3DObjects[targetPlayer] = nil
    end
end

local function createPlayerESP(targetPlayer)
    if targetPlayer == LocalPlayer then return end
    if isTeammate(targetPlayer) and not espShowTeammates then return end
    clearPlayerESP(targetPlayer)
    if not targetPlayer.Character then return end
    local character = targetPlayer.Character
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not rootPart or not humanoid then return end
    local espGroup = {}
    local team = isTeammate(targetPlayer)
    local fillC = team and espTeammateColor or espEnemyColor

    if espBoxes then
        local highlight = Instance.new("Highlight")
        highlight.FillColor = fillC; highlight.OutlineColor = Color3.new(1, 1, 1)
        highlight.FillTransparency = espFillTransparency; highlight.Adornee = character; highlight.Parent = character
        espGroup.highlight = highlight
    end

    if espNames or espHealth or espDistance then
        local bb = Instance.new("BillboardGui")
        bb.Name = "NOX_ESP_" .. targetPlayer.Name; bb.AlwaysOnTop = true
        bb.ExtentsOffset = Vector3.new(0, 3, 0); bb.Size = UDim2.new(0, 200, 0, 50)
        bb.Adornee = rootPart; bb.Parent = rootPart
        local textLabel = Instance.new("TextLabel")
        textLabel.BackgroundTransparency = 1; textLabel.Size = UDim2.new(1, 0, 1, 0)
        textLabel.TextColor3 = fillC; textLabel.TextSize = espTextSize
        textLabel.Font = Enum.Font.Code; textLabel.TextStrokeTransparency = 0
        textLabel.TextStrokeColor3 = Color3.new(0, 0, 0); textLabel.Parent = bb
        espGroup.billboard = bb; espGroup.textLabel = textLabel
    end

    if espTracers and (not team or espShowTeammates) then
        local tracer = Drawing.new("Line")
        tracer.Visible = false; tracer.Color = espTracerColor
        tracer.Thickness = espTracerThickness; tracer.Transparency = 1
        tracerLines[targetPlayer] = tracer
    end

    if espBox3D then
        local boxLines = {}
        for i = 1, 12 do
            local line = Drawing.new("Line")
            line.Visible = false; line.Color = espBox3DColor
            line.Thickness = espBox3DThickness; line.Transparency = 1
            table.insert(boxLines, line)
        end
        box3DObjects[targetPlayer] = boxLines
    end

    espObjects[targetPlayer] = espGroup
end

updateESP = function()
    if not espEnabled then clearESP(); return end
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then createPlayerESP(plr) end
    end
    for plr, _ in pairs(espObjects) do
        if not Players:FindFirstChild(plr) then clearPlayerESP(plr) end
    end
end

updateESPHp = function()
    local cam = Workspace.CurrentCamera
    if not cam or not RootPart or not RootPart.Parent then return end
    for targetPlayer, espGroup in pairs(espObjects) do
        if targetPlayer.Character then
            local character = targetPlayer.Character
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            local rootPart = character:FindFirstChild("HumanoidRootPart")
            if rootPart and espGroup.textLabel and espGroup.textLabel.Parent then
                local isDead = humanoid and humanoid.Health <= 0
                local team = isTeammate(targetPlayer)
                local text = ""
                if espNames then text = text .. targetPlayer.Name .. (isDead and " [DEAD]" or "") .. "\n" end
                if not isDead or espShowDead then
                    if espHealth and humanoid then
                        text = text .. (isDead and "DEAD\n" or math.floor(humanoid.Health) .. "/" .. math.floor(humanoid.MaxHealth) .. "\n")
                    end
                    if espDistance and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                        local dist = math.floor((LocalPlayer.Character.HumanoidRootPart.Position - rootPart.Position).Magnitude)
                        text = text .. dist .. "m"
                    end
                end
                local fillColor = isDead and espDeadColor or (team and espTeammateColor or espEnemyColor)
                espGroup.textLabel.Text = text; espGroup.textLabel.TextColor3 = fillColor; espGroup.textLabel.TextSize = espTextSize
                if espGroup.highlight and espGroup.highlight.Parent then espGroup.highlight.FillColor = fillColor end
            end
        end
    end

    local camera = Workspace.CurrentCamera
    for targetPlayer, tracer in pairs(tracerLines) do
        if targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local humanoid = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
            local isDead = humanoid and humanoid.Health <= 0
            if isDead and not espShowDead then tracer.Visible = false; continue end
            local screenPos, onScreen = camera:WorldToViewportPoint(targetPlayer.Character.HumanoidRootPart.Position)
            if onScreen then
                tracer.From = Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y)
                tracer.To = Vector2.new(screenPos.X, screenPos.Y); tracer.Visible = true
                tracer.Color = espTracerColor; tracer.Thickness = espTracerThickness
            else tracer.Visible = false end
        else tracer.Visible = false end
    end

    for targetPlayer, lines in pairs(box3DObjects) do
        if targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local humanoid = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
            local rootPart = targetPlayer.Character.HumanoidRootPart
            if not humanoid or not rootPart then for _, l in ipairs(lines) do l.Visible = false end; continue end
            local isDead = humanoid.Health <= 0
            if isDead and not espShowDead then for _, l in ipairs(lines) do l.Visible = false end; continue end
            local size = Vector3.new(3, 5, 3)
            local corners = {
                rootPart.Position + Vector3.new(-size.X/2, -size.Y/2, -size.Z/2),
                rootPart.Position + Vector3.new(size.X/2, -size.Y/2, -size.Z/2),
                rootPart.Position + Vector3.new(size.X/2, size.Y/2, -size.Z/2),
                rootPart.Position + Vector3.new(-size.X/2, size.Y/2, -size.Z/2),
                rootPart.Position + Vector3.new(-size.X/2, -size.Y/2, size.Z/2),
                rootPart.Position + Vector3.new(size.X/2, -size.Y/2, size.Z/2),
                rootPart.Position + Vector3.new(size.X/2, size.Y/2, size.Z/2),
                rootPart.Position + Vector3.new(-size.X/2, size.Y/2, size.Z/2)
            }
            local screenCorners = {}; local allVisible = true
            for i, corner in ipairs(corners) do
                local sp, vis = camera:WorldToViewportPoint(corner)
                screenCorners[i] = Vector2.new(sp.X, sp.Y)
                if not vis then allVisible = false end
            end
            if not allVisible then for _, l in ipairs(lines) do l.Visible = false end; continue end
            local connections = {{1,2},{2,3},{3,4},{4,1},{5,6},{6,7},{7,8},{8,5},{1,5},{2,6},{3,7},{4,8}}
            for i, conn in ipairs(connections) do
                if lines[i] then
                    lines[i].From = screenCorners[conn[1]]; lines[i].To = screenCorners[conn[2]]
                    lines[i].Visible = true; lines[i].Color = isDead and espDeadColor or espBox3DColor
                    lines[i].Thickness = espBox3DThickness
                end
            end
        else for _, l in ipairs(lines) do l.Visible = false end end
    end
end

local function stripCharacter(char, isLocal)
    pcall(function()
        if not isLocal then
            for _, v in pairs(char:GetDescendants()) do
                if v:IsA("Accessory") or v:IsA("Hat") or v:IsA("HairAccessory") or v:IsA("FaceAccessory") or v:IsA("NeckAccessory") or v:IsA("ShouldersAccessory") or v:IsA("FrontAccessory") or v:IsA("BackAccessory") or v:IsA("WaistAccessory") then v:Destroy()
                elseif v:IsA("Shirt") or v:IsA("Pants") or v:IsA("ShirtGraphic") or v:IsA("BodyColors") or v:IsA("CharacterMesh") then v:Destroy()
                elseif v:IsA("Decal") or v:IsA("Texture") or v:IsA("SurfaceAppearance") then v:Destroy()
                elseif v:IsA("SpecialMesh") then v:Destroy()
                elseif v:IsA("MeshPart") then v.TextureID = ""
                elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam") or v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") then v:Destroy()
                elseif v:IsA("PointLight") or v:IsA("SpotLight") or v:IsA("SurfaceLight") then v:Destroy()
                elseif v:IsA("Sound") then v.Volume = 0; v.Playing = false
                end
            end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then pcall(function() hum:SetStateEnabled(Enum.HumanoidStateType.Physics, false) end) end
        end
    end)
end

local function deepStripWorkspace()
    pcall(function()
        for _, v in pairs(Workspace:GetDescendants()) do
            if v:IsA("Texture") or v:IsA("Decal") or v:IsA("SurfaceAppearance") then v:Destroy()
            elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam") or v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") then v:Destroy()
            elseif v:IsA("Sound") then v.Volume = 0; v.Playing = false
            elseif v:IsA("PointLight") or v:IsA("SpotLight") or v:IsA("SurfaceLight") then v:Destroy()
            elseif v:IsA("SpecialMesh") then v:Destroy()
            elseif v:IsA("MeshPart") then v.TextureID = ""; pcall(function() v.RenderFidelity = Enum.RenderFidelity.Performance end)
            elseif v:IsA("BlockMesh") or v:IsA("CylinderMesh") or v:IsA("DataStoreMesh") then v.TextureId = ""
            elseif v:IsA("UnionOperation") or v:IsA("NegateOperation") or v:IsA("IntersectOperation") then pcall(function() v.RenderFidelity = Enum.RenderFidelity.Performance end)
            elseif v:IsA("Shirt") or v:IsA("Pants") or v:IsA("ShirtGraphic") or v:IsA("BodyColors") or v:IsA("CharacterMesh") then v:Destroy()
            elseif v:IsA("Accessory") or v:IsA("Hat") or v:IsA("HairAccessory") or v:IsA("FaceAccessory") or v:IsA("NeckAccessory") or v:IsA("ShouldersAccessory") or v:IsA("FrontAccessory") or v:IsA("BackAccessory") or v:IsA("WaistAccessory") then v:Destroy()
            elseif v:IsA("Atmosphere") or v:IsA("Sky") or v:IsA("Clouds") then v:Destroy()
            elseif v:IsA("PostEffect") then v.Enabled = false
            end
        end
    end)
    pcall(function()
        Workspace.Terrain.WaterWaveSize = 0
        Workspace.Terrain.WaterWaveSpeed = 0
        Workspace.Terrain.WaterReflectance = 0
        Workspace.Terrain.WaterTransparency = 1
        Workspace.Terrain.Decoration = false
        Workspace.Terrain.ExplicitMaterial = Enum.Material.Plastic
    end)
end

updateESP = function()
    local anyESP = espEnabled or distanceESPEnabled
    if not anyESP then clearESP(); return end
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local head = plr.Character:FindFirstChild("Head")
            local root = plr.Character:FindFirstChild("HumanoidRootPart")
            local hum = plr.Character:FindFirstChild("Humanoid")
            if head and root and hum then
                if not espObjects[plr.Name] then
                    local bb = Instance.new("BillboardGui")
                    bb.Size = UDim2.new(0, 140, 0, 50)
                    bb.StudsOffset = Vector3.new(0, 2.5, 0)
                    bb.AlwaysOnTop = true
                    bb.Adornee = head
                    bb.Parent = OverlayFrame
                    local nameTag = Instance.new("TextLabel")
                    nameTag.Size = UDim2.new(1, 0, 0.5, 0)
                    nameTag.BackgroundTransparency = 1
                    nameTag.Text = plr.Name
                    nameTag.TextColor3 = Color3.fromRGB(255, 0, 0)
                    nameTag.TextStrokeTransparency = 0
                    nameTag.TextScaled = true
                    nameTag.Font = Enum.Font.Code
                    nameTag.Parent = bb
                    local hpTag = Instance.new("TextLabel")
                    hpTag.Size = UDim2.new(1, 0, 0.5, 0)
                    hpTag.Position = UDim2.new(0, 0, 0.5, 0)
                    hpTag.BackgroundTransparency = 1
                    hpTag.Text = ""
                    hpTag.TextColor3 = Color3.fromRGB(0, 200, 255)
                    hpTag.TextStrokeTransparency = 0
                    hpTag.TextScaled = true
                    hpTag.Font = Enum.Font.Code
                    hpTag.Parent = bb
                    espObjects[plr.Name] = {billboard = bb, hpLabel = hpTag, nameTag = nameTag}
                end
                local data = espObjects[plr.Name]
                data.hpLabel.Text = math.floor(hum.Health) .. "/" .. math.floor(hum.MaxHealth)
                local dist = math.floor((root.Position - RootPart.Position).Magnitude)
                if espEnabled and distanceESPEnabled then
                    data.nameTag.Text = plr.Name .. " [" .. dist .. "m]"
                elseif espEnabled then
                    data.nameTag.Text = plr.Name
                elseif distanceESPEnabled then
                    data.nameTag.Text = "[" .. dist .. "m]"
                end
            end
        end
    end
    for name, data in pairs(espObjects) do
        if not Players:FindFirstChild(name) then
            if data.billboard then data.billboard:Destroy() end
            espObjects[name] = nil
        end
    end
end

updateESPHp = function()
    for name, data in pairs(espObjects) do
        local plr = Players:FindFirstChild(name)
        if plr and plr.Character then
            local hum = plr.Character:FindFirstChild("Humanoid")
            if hum then data.hpLabel.Text = math.floor(hum.Health) .. "/" .. math.floor(hum.MaxHealth) end
        end
    end
end

local function applyReduxPreset(cfg)
    if cfg.bloom then
        reduxBloomEnabled = cfg.bloom.on
        reduxBloomIntensity = cfg.bloom.intensity or 1
        reduxBloomSize = cfg.bloom.size or 24
        reduxBloomThreshold = cfg.bloom.threshold or 1
        if reduxBloomEnabled then
            if not bloomInstance then bloomInstance = Instance.new("BloomEffect") end
            bloomInstance.Intensity = reduxBloomIntensity; bloomInstance.Size = reduxBloomSize; bloomInstance.Threshold = reduxBloomThreshold; bloomInstance.Parent = Lighting
        else
            if bloomInstance then bloomInstance:Destroy(); bloomInstance = nil end
        end
    end
    if cfg.cc then
        reduxCCEnabled = cfg.cc.on
        reduxCCBrightness = cfg.cc.brightness or 0; reduxCCContrast = cfg.cc.contrast or 0; reduxCCSaturation = cfg.cc.saturation or 0
        if reduxCCEnabled then
            if not ccInstance then ccInstance = Instance.new("ColorCorrectionEffect") end
            ccInstance.Brightness = reduxCCBrightness; ccInstance.Contrast = reduxCCContrast; ccInstance.Saturation = reduxCCSaturation; ccInstance.Parent = Lighting
        else
            if ccInstance then ccInstance:Destroy(); ccInstance = nil end
        end
    end
    if cfg.sun then
        reduxSunRaysEnabled = cfg.sun.on; reduxSunRaysIntensity = cfg.sun.intensity or 0.1; reduxSunRaysSpread = cfg.sun.spread or 0.5
        if reduxSunRaysEnabled then
            if not sunRaysInstance then sunRaysInstance = Instance.new("SunRaysEffect") end
            sunRaysInstance.Intensity = reduxSunRaysIntensity; sunRaysInstance.Spread = reduxSunRaysSpread; sunRaysInstance.Parent = Lighting
        else
            if sunRaysInstance then sunRaysInstance:Destroy(); sunRaysInstance = nil end
        end
    end
    if cfg.blur then
        reduxBlurEnabled = cfg.blur.on; reduxBlurSize = cfg.blur.size or 5
        if reduxBlurEnabled then
            if not blurInstance then blurInstance = Instance.new("BlurEffect") end
            blurInstance.Size = reduxBlurSize; blurInstance.Parent = Lighting
        else
            if blurInstance then blurInstance:Destroy(); blurInstance = nil end
        end
    end
    if cfg.fog then
        reduxFogColorEnabled = cfg.fog.on; reduxFogColorR = cfg.fog.r or 199; reduxFogColorG = cfg.fog.g or 199; reduxFogColorB = cfg.fog.b or 199

        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0, 1, 0) end
        if dir.Magnitude > 0 then RootPart.Velocity = dir.Unit * flySpeed * 10
        else RootPart.Velocity = Vector3.new(0, RootPart.Velocity.Y, 0) end
    end
    if walkSpeedEnabled then Humanoid.WalkSpeed = walkSpeedValue end
    if noclipEnabled then for _, p in pairs(Character:GetDescendants()) do if p:IsA("BasePart") then p.CanCollide = false end end end
    if bhopEnabled and Humanoid:GetState() ~= Enum.HumanoidStateType.Running then
        local moveDir = Vector3.new(Camera.CFrame.LookVector.X, 0, Camera.CFrame.LookVector.Z)
        if moveDir.Magnitude > 0 then RootPart.Velocity = RootPart.Velocity + moveDir.Unit * bhopSpeed end
    end
    if infJumpEnabled and UserInputService:IsKeyDown(Enum.KeyCode.Space) then
        Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
        RootPart.Velocity = Vector3.new(RootPart.Velocity.X, jumpPower, RootPart.Velocity.Z)
    end
    if hitboxExpanderEnabled then
        for _, plr in pairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                for _, part in pairs(plr.Character:GetChildren()) do
                    if part:IsA("BasePart") then
                        local key = plr.Name .. part.Name
                        if not originalSizes[key] then originalSizes[key] = part.Size end
                        part.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
                    end
                end
            end
        end
    end
    if fakeLagEnabled and math.random() < 0.3 then RootPart.Velocity = Vector3.new(RootPart.Velocity.X * 0.1, RootPart.Velocity.Y, RootPart.Velocity.Z * 0.1) end
    local finalAngle = 0
    local hasRotation = false
    if spinbotEnabled then spinAngle = (spinAngle + spinSpeed * dt * 360) % 360; finalAngle = spinAngle; hasRotation = true end
    if antiAimEnabled then
        local jitter = math.random(-40, 40)
        antiAimAngle = (antiAimAngle + (180 + jitter) * dt) % 360
        finalAngle = finalAngle + antiAimAngle; hasRotation = true
    end
    if hasRotation then RootPart.CFrame = CFrame.new(RootPart.CFrame.Position) * CFrame.Angles(0, math.rad(finalAngle), 0) end
    if aimbotEnabled then
        local closest, closestDist = nil, aimbotFOV
        local mousePos = UserInputService:GetMouseLocation()
        for _, plr in pairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                if aimbotTeamCheck and plr.Team == LocalPlayer.Team then continue end
                local head = plr.Character:FindFirstChild("Head")
                local hum = plr.Character:FindFirstChild("Humanoid")
                if head and hum and hum.Health > 0 then
                    if aimbotWallCheck then
                        local rayParams = RaycastParams.new()
                        rayParams.FilterDescendantsInstances = {Character, plr.Character}
                        rayParams.FilterType = Enum.RaycastFilterType.Exclude
                        if Workspace:Raycast(Camera.CFrame.Position, head.Position - Camera.CFrame.Position, rayParams) then continue end
                    end
                    local rp, onScreen = Camera:WorldToViewportPoint(head.Position)
                    if onScreen then
                        local screenDist = (Vector2.new(rp.X, rp.Y) - mousePos).Magnitude
                        if screenDist < closestDist then closestDist = screenDist; closest = head end
                    end
                end
            end
        end
        if closest then
            local dir = (closest.Position - Camera.CFrame.Position).Unit
            local goal = Camera.CFrame.LookVector:Lerp(dir, 1 / aimbotSmoothing)
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, Camera.CFrame.Position + goal)
        end
    end
    if silentAimEnabled then
        local closest, closestDist = nil, aimbotFOV
        local mousePos = UserInputService:GetMouseLocation()
        for _, plr in pairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                if aimbotTeamCheck and plr.Team == LocalPlayer.Team then continue end
                local head = plr.Character:FindFirstChild("Head")
                local hum = plr.Character:FindFirstChild("Humanoid")
                if head and hum and hum.Health > 0 then
                    if aimbotWallCheck then
                        local rayParams = RaycastParams.new()
                        rayParams.FilterDescendantsInstances = {Character, plr.Character}
                        rayParams.FilterType = Enum.RaycastFilterType.Exclude
                        if Workspace:Raycast(Camera.CFrame.Position, head.Position - Camera.CFrame.Position, rayParams) then continue end
                    end
                    local rp, onScreen = Camera:WorldToViewportPoint(head.Position)
                    if onScreen then
                        local screenDist = (Vector2.new(rp.X, rp.Y) - mousePos).Magnitude
                        if screenDist < closestDist then closestDist = screenDist; closest = head end
                    end
                end
            end
        end
        if closest then
            pcall(function()
                local mouse = LocalPlayer:GetMouse()
                mouse.Target = closest
                mouse.TargetFilter = Character
            end)
        end
    end
    if flingEnabled then
        flingSpinAngle = (flingSpinAngle + 20 * dt * 360) % 360
        RootPart.CFrame = CFrame.new(RootPart.CFrame.Position) * CFrame.Angles(0, 0, math.rad(flingSpinAngle))
        RootPart.RotVelocity = Vector3.new(0, 9999, 0)
    end
    if noFogEnabled then pcall(function() Lighting.FogStart = 0; Lighting.FogEnd = 999999 end) end
    if customFOVEnabled then Camera.FieldOfView = customFOVValue elseif Camera.FieldOfView ~= 70 then Camera.FieldOfView = 70 end
    if stretchEnabled and origViewportSize then
        pcall(function() local cam = Workspace.CurrentCamera; if cam then cam.ViewportSize = Vector2.new(origViewportSize.X / stretchX, origViewportSize.Y / stretchY) end end)
    end
end)
